import "date"

from(bucket:  "classification_test")
  |> range(start: -1h) // adjust your time range as needed
  |> filter(fn: (r) => r._measurement == "classification")
  |> filter(fn: (r) => r._field == "event_timestamp")
  |> map(fn: (r) => ({
      r with
      event_time_ns: int(v: r._value) * 1000000,                                   // convert ms → ns
      time_ns: int(v: uint(v: r._time)),                                           // convert Flux time → int ns
      delay_ms: float(v: (int(v: uint(v: r._time)) - int(v: r._value) * 1000000)) / 1000000.0  // ns diff → ms
  }))
  |> keep(columns: ["_time", "sensor_id", "building", "delay_ms"])


import "date"

from(bucket: "sound")
  |> range(start: v.timeRangeStart, stop: v.timeRangeStop)
  |> filter(fn: (r) => r["sensor_id"] == v.linux_host)
  |> filter(fn: (r) => r._measurement == "classification")
  |> filter(fn: (r) => r._field == "event_timestamp")
  |> map(fn: (r) => ({
      r with
      event_time_ns: int(v: r._value) * 1000000,                                   // convert ms → ns
      time_ns: int(v: uint(v: r._time)),                                           // convert Flux time → int ns
      delay_ms: float(v: (int(v: uint(v: r._time)) - int(v: r._value) * 1000000)) / 1000000.0  // ns diff → ms
  }))
  |> keep(columns: ["delay_ms"])
  |> aggregateWindow(every: v.windowPeriod, fn: mean, createEmpty: false)
  |> yield(name: "mean")
